      เปิด / ปิด
      1. เปิดหน้าต่างรัน SQL ที่ต่อฐานข้อมูล HOSxP XE (PostgreSQL) ได้ เช่น SQL Query ใน HOSxP XE, pgAdmin หรือ Navicat ใช้สิทธิอ่านอย่างเดียวก็พอ
      2. ตรวจค่าที่ต้องตั้งก่อนใช้ครั้งแรก รันคำสั่งในกล่อง “ตรวจสอบก่อนใช้” ท้ายโค้ด แล้วแก้ selfpay_paidst ให้ตรงกับรหัส paidst ที่หมายถึงผู้ป่วยชำระเอง (ค่าเริ่มต้น 01, 03) และดูว่ารายการ ADP type 2, 3, 5 ผูกไว้ครบ
      3. แก้ช่วงวันที่ ที่บรรทัด d1 และ d2 ให้ตรงกับเดือนของ STM ที่จะเทียบ เช่น STM งวด 6908 ใช้ 2026-08-01 ถึง 2026-08-31
      4. รันคำสั่งหลัก (ตั้งแต่ WITH params ถึง ORDER BY) แล้วส่งออกผลเป็น Excel (.xls หรือ .xlsx) โดยให้แถวแรกเป็นชื่อคอลัมน์ ห้ามเปลี่ยนชื่อหรือลำดับคอลัมน์
      5. แนบไฟล์ในช่อง 1 ด้านบน ระบบใช้ vn กับ cid + vstdate จับคู่ และใช้ยอด debtor_inst ในคอลัมน์ M เป็นยอดลูกหนี้

      เงื่อนไขที่โค้ดคัดเข้า

      • สิทธิ 03, 05, 31 เข้าทุกราย
      • สิทธิ UC อื่น (UCS, WEL) เข้าเมื่อมีรายการผูก ADP type 2 ทุกรหัส, type 3 เฉพาะ CONS01 EVA001 STEMI1 STROK1 หรือ type 5 ทุกรหัส
      • สิทธิ 43, 04 ถ้ามีแค่ ADP type 5 ไม่นับ

      คอลัมน์ที่ได้

      • debtor_visit (L) ยอดทั้ง visit หักส่วนชำระเอง
      • debtor_inst (M) ยอดเฉพาะ ADP type 2 ส่วนสิทธิ 05, 31 เป็นยอดทั้ง visit
      • adp_items รายการ ADP ที่พบ เช่น T2:8307, T5:WALKIN
      opcr_query.sql · PostgreSQL 
      /* =====================================================================
   ลูกหนี้ UC บริการกรณีเฉพาะ (ลูกหนี้รายตัว OPD) — HOSxP XE / PostgreSQL

   เงื่อนไขเข้า:
     (ก) สิทธิ 03 (สิทธิว่าง), 05, 31 (บัตรทองต่างจังหวัด)  -> ทุกราย
     (ข) สิทธิ UC อื่น (hipdata UCS, WEL) ที่มีรายการ (ทั้ง drug และ non-drug) ผูก ADP:
           - Type 2 Instrument            : ทุกรหัส
           - Type 3 ค่าบริการอื่นๆ         : เฉพาะ Cons01, EVA001, STEMI1, STROK1
           - Type 5 Project code          : ทุกรหัส
         ยกเว้นสิทธิ 43, 04 (บัตรทองนอกเขตในจังหวัด):
           ถ้ามีแค่ ADP type 5 ไม่นับ / ต้องมี ADP type 2 หรือ type 3 ข้างต้นด้วยจึงนับ

   ผลลัพธ์:
     debtor_visit = ยอดทั้ง visit หักส่วนผู้ป่วยชำระเอง
     debtor_inst  = ยอดเฉพาะ ADP type 2 (สิทธิ 05, 31 = ยอดทั้ง visit)
   ===================================================================== */

WITH params AS (
    SELECT
        DATE '2026-10-01'                    AS d1,              -- << วันที่เริ่ม
        DATE '2026-10-31'                    AS d2,              -- << วันที่สิ้นสุด
        ARRAY['03','05','31']::text[]        AS always_pttype,   -- นับทุกราย
        ARRAY['05','31']::text[]             AS fullvisit_pttype,-- ช่อง 2 = ยอดทั้ง visit
        ARRAY['43','04']::text[]             AS in_prov_pttype,  -- ต้องมี ADP ที่ไม่ใช่ type 5
        ARRAY['UCS','WEL']::text[]           AS uc_hipdata,
        ARRAY['CONS01','EVA001','STEMI1','STROK1']::text[] AS adp3_codes,
        ARRAY['01','03']::text[]             AS selfpay_paidst   -- << ตรวจด้วย SELECT * FROM paidst;
),

/* visit OPD ที่สิทธิเข้าข่าย */
v AS (
    SELECT ov.vn, ov.hn, ov.vstdate, ov.pttype,
           pt.name          AS pttype_name,
           pt.hipdata_code,
           (ov.pttype = ANY(p.always_pttype))    AS is_always,
           (ov.pttype = ANY(p.fullvisit_pttype)) AS is_fullvisit,
           (ov.pttype = ANY(p.in_prov_pttype))   AS is_in_prov
    FROM ovst ov
    JOIN pttype pt ON pt.pttype = ov.pttype
    CROSS JOIN params p
    WHERE ov.vstdate BETWEEN p.d1 AND p.d2
      AND COALESCE(ov.an,'') = ''                          -- เฉพาะ OPD
      AND ( ov.pttype = ANY(p.always_pttype)
            OR pt.hipdata_code = ANY(p.uc_hipdata) )
),

/* รายการค่ารักษา + ADP จาก drugitems / nondrugitems */
items AS (
    SELECT op.vn,
           op.icode,
           COALESCE(op.sum_price,0) AS sum_price,
           (COALESCE(op.paidst,'') = ANY(p.selfpay_paidst)) AS is_selfpay,
           COALESCE(n.nhso_adp_type_id, d.nhso_adp_type_id)          AS adp_type,
           UPPER(TRIM(COALESCE(n.nhso_adp_code, d.nhso_adp_code)))    AS adp_code
    FROM opitemrece op
    JOIN v ON v.vn = op.vn
    LEFT JOIN nondrugitems n ON n.icode = op.icode
    LEFT JOIN drugitems    d ON d.icode = op.icode
    CROSS JOIN params p
),

items2 AS (
    SELECT i.*,
           CASE
             WHEN i.adp_type = 2 THEN 'T2'
             WHEN i.adp_type = 3 AND i.adp_code = ANY(p.adp3_codes) THEN 'T3'
             WHEN i.adp_type = 5 THEN 'T5'
           END AS adp_group
    FROM items i CROSS JOIN params p
),

agg AS (
    SELECT vn,
           SUM(sum_price)                                                     AS total_amount,
           SUM(sum_price) FILTER (WHERE is_selfpay)                           AS selfpay_amount,
           SUM(sum_price) FILTER (WHERE adp_group = 'T2' AND NOT is_selfpay)  AS inst_amount,
           COUNT(*) FILTER (WHERE adp_group = 'T2')                           AS n_t2,
           COUNT(*) FILTER (WHERE adp_group = 'T3')                           AS n_t3,
           COUNT(*) FILTER (WHERE adp_group = 'T5')                           AS n_t5,
           STRING_AGG(DISTINCT adp_group || ':' || COALESCE(adp_code, icode), ', ')
                 FILTER (WHERE adp_group IS NOT NULL)                         AS adp_items
    FROM items2
    GROUP BY vn
)

SELECT
    v.vstdate,
    v.vn,
    v.hn,
    pa.cid,
    CONCAT(pa.pname, pa.fname, ' ', pa.lname)          AS patient_name,
    v.pttype,
    v.pttype_name,
    v.hipdata_code,
    vs.pdx,
    a.total_amount,
    COALESCE(a.selfpay_amount,0)                        AS selfpay_amount,
    a.total_amount - COALESCE(a.selfpay_amount,0)       AS debtor_visit,   -- ช่อง 1
    CASE WHEN v.is_fullvisit
         THEN a.total_amount - COALESCE(a.selfpay_amount,0)
         ELSE COALESCE(a.inst_amount,0)
    END                                                  AS debtor_inst,    -- ช่อง 2
    a.n_t2, a.n_t3, a.n_t5,
    a.adp_items,
    vs.paid_money                                        AS vn_stat_paid_money -- เทียบกับ selfpay_amount
FROM v
JOIN agg a           ON a.vn  = v.vn
JOIN patient pa      ON pa.hn = v.hn
LEFT JOIN vn_stat vs ON vs.vn = v.vn
WHERE v.is_always
   OR (v.is_in_prov     AND (a.n_t2 + a.n_t3) > 0)              -- 43/04: มีแค่ T5 ไม่นับ
   OR (NOT v.is_in_prov AND (a.n_t2 + a.n_t3 + a.n_t5) > 0)
ORDER BY v.vstdate, v.vn;


/* ---------------- ตรวจสอบก่อนใช้ ----------------
-- 1) paidst ที่หมายถึง "ผู้ป่วยชำระเอง" (แก้ selfpay_paidst ให้ตรง)
SELECT * FROM paidst ORDER BY paidst;

-- 2) ชื่อ ADP type
SELECT * FROM nhso_adp_type ORDER BY nhso_adp_type_id;

-- 3) รายการที่ผูก ADP ไว้ (ยา + ไม่ใช่ยา)
SELECT 'nondrug' src, icode, name, nhso_adp_type_id, nhso_adp_code
FROM nondrugitems WHERE nhso_adp_type_id IN (2,3,5)
UNION ALL
SELECT 'drug', icode, name, nhso_adp_type_id, nhso_adp_code
FROM drugitems WHERE nhso_adp_type_id IN (2,3,5)
ORDER BY 4, 3;
-------------------------------------------------- */
